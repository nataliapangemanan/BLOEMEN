def _tokenize(text: str) -> list[str]:
    # Tokenisasi Bahasa Indonesia: Stopword Removal -> Stemming Morfologi ->
Alfanumerik
    cleaned = _stopword_remover.remove(text)
    stemmed = _stemmer.stem(cleaned)
    tokens = re.findall(r"\b\w+\b", stemmed.lower())
    return [t for t in tokens if len(t) >= BM25_MIN_TOKEN_LEN]
def retrieve_chunks_bm25(query: str) -> list[dict]:
    bm25, chunks = _load_index()

**5. Backend FastAPI (app.py)** 

```
from fastapi import FastAPI
from fastapi.responses import StreamingResponse
from pydantic import BaseModel
from src.chain import get_response
from src.config import LLM_MODEL_NAME
from src.logger_manager import get_logger
logger = get_logger("app")
app = FastAPI(title="UNSRAT RAG Chatbot API", version="1.0.0")
class ChatRequest(BaseModel):
    query: str
    config: str = "b"
    chat_history: list[dict] = []
    model: str = LLM_MODEL_NAME
@app.post("/api/chat")
async def chat(request: ChatRequest):
    logger.info(
        f"POST /api/chat | config={request.config} | model={request.model} | "
        f"query='{request.query[:60]}'"
    )
    def event_generator():
        yield from get_response(
            query=request.query,
            config=request.config,
            chat_history=request.chat_history,
            model_name=request.model,
            streaming=True,
        )
    return StreamingResponse(
        event_generator(),
        media_type="text/event-stream",
        headers={
            "Cache-Control": "no-cache",
            "X-Accel-Buffering": "no",
        },
    )
```

**Penjelasan Kode:** Endpoint '/api/chat' melayani permintaan percakapan dari antarmuka web dan mengembalikan aliran respons Server-Sent Events (SSE) secara asinkron.
